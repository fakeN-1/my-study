<h3>普通文本框input</h3>
plaxehoder属性:透明的提示词文本<br>
size:控制输入框的长度<br>
maxlebght属性:限制最大的字数<br>
submit属性:按钮

```html
    <div>请输入需要搜索的内容<br>
    <input type="text" plaxehoder="请输入需内容" size="20" maxlenght="10" placeholder="请输入需内容">
    </div>
```

  <main>
    <h1>HTML 表单基础学习</h1>

    <!-- 普通文本框：type="text" 用于输入一行普通文字。 -->

  <script>
    // 监听演示表单的提交事件，阻止页面刷新并读取输入值。
    const demoForm = document.querySelector("#demo-form");
    const result = document.querySelector("#submit-result");
    demoForm.addEventListener("submit", (event) => {
      event.preventDefault(); // 取消浏览器默认提交，方便在当前页面查看结果。

    // 拖动范围滑块时，同步更新旁边显示的数字。
    const range = document.querySelector("#satisfaction");
